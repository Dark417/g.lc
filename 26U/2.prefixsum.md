#### 面试题 17.24. [最大子矩阵](https://leetcode.com/problems/max-submatrix-lcci/) [H]

Return the coordinates of a maximum-sum rectangular submatrix.  
`array` `dynamic-programming` `matrix` `prefix-sum`

##### Approach 1: Row Prefix Sums + Kadane's Algorithm

Fix the top and bottom rows. Column prefix sums collapse that row band into a one-dimensional array, where Kadane's algorithm finds the best column interval. This checks every rectangle family without enumerating both column boundaries.

```python
class Solution:
    def getMaxMatrix(self, matrix: list[list[int]]) -> list[int]:
        rows, cols = len(matrix), len(matrix[0])
        column_prefix = [[0] * cols for _ in range(rows + 1)]

        for r in range(rows):
            for c in range(cols):
                column_prefix[r + 1][c] = column_prefix[r][c] + matrix[r][c]

        best_sum = float("-inf")
        answer = [0, 0, 0, 0]

        for top in range(rows):
            for bottom in range(top, rows):
                current_sum = 0
                left = 0

                for right in range(cols):
                    value = column_prefix[bottom + 1][right] - column_prefix[top][right]
                    if current_sum < 0:
                        current_sum = value
                        left = right
                    else:
                        current_sum += value

                    if current_sum > best_sum:
                        best_sum = current_sum
                        answer = [top, left, bottom, right]

        return answer
# Time: O(rows^2 * cols), Space: O(rows * cols)
```

#### 3721. [最长平衡子数组 II](https://leetcode.com/problems/longest-balanced-subarray-ii/) [H]

Find the longest subarray containing equally many distinct even and distinct odd values.  
`segment-tree` `array` `hash-table` `divide-and-conquer` `prefix-sum`

##### Approach 1: Distinct-Parity Prefix Balance + Lazy Segment Tree

For every possible left boundary, maintain `distinct_odd - distinct_even` for the subarray ending at the current right boundary. A value contributes only to left boundaries after its previous occurrence, so each new element becomes one range addition. The segment tree stores minimum and maximum balances and finds the earliest zero balance; adjacent left-boundary balances differ by at most one, so a range containing zero between its minimum and maximum must contain an actual zero.

```python
class Solution:
    def longestBalanced(self, nums: list[int]) -> int:
        n = len(nums)
        minimum = [0] * (4 * n)
        maximum = [0] * (4 * n)
        lazy = [0] * (4 * n)

        def apply(node: int, delta: int) -> None:
            minimum[node] += delta
            maximum[node] += delta
            lazy[node] += delta

        def push(node: int) -> None:
            if lazy[node] != 0:
                apply(node * 2, lazy[node])
                apply(node * 2 + 1, lazy[node])
                lazy[node] = 0

        def range_add(
            node: int,
            segment_left: int,
            segment_right: int,
            query_left: int,
            query_right: int,
            delta: int,
        ) -> None:
            if query_left <= segment_left and segment_right <= query_right:
                apply(node, delta)
                return

            push(node)
            middle = (segment_left + segment_right) // 2
            if query_left <= middle:
                range_add(node * 2, segment_left, middle, query_left, query_right, delta)
            if query_right > middle:
                range_add(node * 2 + 1, middle + 1, segment_right, query_left, query_right, delta)
            minimum[node] = min(minimum[node * 2], minimum[node * 2 + 1])
            maximum[node] = max(maximum[node * 2], maximum[node * 2 + 1])

        def first_zero(node: int, left: int, right: int) -> int:
            if left == right:
                return left

            push(node)
            middle = (left + right) // 2
            left_node = node * 2
            if minimum[left_node] <= 0 <= maximum[left_node]:
                return first_zero(left_node, left, middle)
            return first_zero(left_node + 1, middle + 1, right)

        last_position = {}
        answer = 0

        for right, value in enumerate(nums):
            delta = 1 if value % 2 else -1
            start = last_position.get(value, -1) + 1
            range_add(1, 0, n - 1, start, right, delta)
            last_position[value] = right

            if minimum[1] <= 0 <= maximum[1]:
                left = first_zero(1, 0, n - 1)
                if left <= right:
                    answer = max(answer, right - left + 1)

        return answer
# Time: O(n log n), Space: O(n)
```

#### 410. [分割数组的最大值](https://leetcode.com/problems/split-array-largest-sum/) [H]

Split `nums` into `k` non-empty contiguous parts and minimize the largest part sum.  
`greedy` `array` `binary-search` `dynamic-programming` `prefix-sum`

##### Approach 1: Prefix Sum + Partition DP

Let `dp[parts][end]` be the smallest possible largest sum after splitting the first `end` values into `parts` groups. Try every final cut and obtain its group sum from the prefix array.

```python
class Solution:
    def splitArray(self, nums: list[int], k: int) -> int:
        n = len(nums)
        prefix = [0]
        for num in nums:
            prefix.append(prefix[-1] + num)

        previous = [float("inf")] * (n + 1)
        previous[0] = 0

        for parts in range(1, k + 1):
            current = [float("inf")] * (n + 1)
            for end in range(parts, n + 1):
                for cut in range(parts - 1, end):
                    last_sum = prefix[end] - prefix[cut]
                    current[end] = min(current[end], max(previous[cut], last_sum))
            previous = current

        return previous[n]
# Time: O(k * n^2), Space: O(n)
```

##### Approach 2: Binary Search + Greedy Feasibility

Binary-search the answer between the largest element and the total sum. For a proposed limit, greedily extend each part as far as possible; this uses the fewest parts and therefore decides feasibility.

```python
class Solution:
    def splitArray(self, nums: list[int], k: int) -> int:
        def required_parts(limit: int) -> int:
            parts = 1
            current_sum = 0
            for num in nums:
                if current_sum + num > limit:
                    parts += 1
                    current_sum = 0
                current_sum += num
            return parts

        low, high = max(nums), sum(nums)
        while low < high:
            middle = (low + high) // 2
            if required_parts(middle) <= k:
                high = middle
            else:
                low = middle + 1
        return low
# Time: O(n log(sum(nums))), Space: O(1)
```

#### 3225. [网格图操作后的最大分数](https://leetcode.com/problems/maximum-score-from-grid-operations/) [H]

Choose a black-prefix height for every column to maximize scored white cells adjacent to black cells.  
`array` `dynamic-programming` `matrix` `prefix-sum`

##### Approach 1: Column Prefix Sums + Optimized Height DP

A column with black height `self_height` scores rows from that height up to the taller adjacent black height. Column prefix sums evaluate this contribution in O(1). The DP remembers the last two heights; prefix and suffix maxima over the older height reduce each transition layer from O(n³) to O(n²).

```python
class Solution:
    def maximumScore(self, grid: list[list[int]]) -> int:
        n = len(grid)
        heights = n + 1
        negative_infinity = -10**30

        column_prefix = [[0] * heights for _ in range(n)]
        for column in range(n):
            for height in range(1, heights):
                column_prefix[column][height] = (
                    column_prefix[column][height - 1] + grid[height - 1][column]
                )

        dp = [[negative_infinity] * heights for _ in range(heights)]
        for first_height in range(heights):
            dp[0][first_height] = 0

        for column in range(1, n):
            next_dp = [[negative_infinity] * heights for _ in range(heights)]
            prefix = column_prefix[column - 1]

            for previous_height in range(heights):
                prefix_best = [negative_infinity] * heights
                best = negative_infinity
                for older_height in range(heights):
                    best = max(best, dp[older_height][previous_height])
                    prefix_best[older_height] = best

                suffix_best = [negative_infinity] * (heights + 1)
                best = negative_infinity
                for older_height in range(heights - 1, -1, -1):
                    contribution = max(
                        0,
                        prefix[older_height] - prefix[previous_height],
                    )
                    best = max(
                        best,
                        dp[older_height][previous_height] + contribution,
                    )
                    suffix_best[older_height] = best

                for current_height in range(heights):
                    contribution = max(
                        0,
                        prefix[current_height] - prefix[previous_height],
                    )
                    next_dp[previous_height][current_height] = max(
                        prefix_best[current_height] + contribution,
                        suffix_best[current_height + 1],
                    )

            dp = next_dp

        answer = 0
        last_prefix = column_prefix[-1]
        for older_height in range(heights):
            for last_height in range(heights):
                final_contribution = max(
                    0,
                    last_prefix[older_height] - last_prefix[last_height],
                )
                answer = max(
                    answer,
                    dp[older_height][last_height] + final_contribution,
                )
        return answer
# Time: O(n^3), Space: O(n^2)
```

#### 862. [和至少为 K 的最短子数组](https://leetcode.com/problems/shortest-subarray-with-sum-at-least-k/) [H]

Return the length of the shortest non-empty subarray whose sum is at least `k`.  
`queue` `array` `binary-search` `prefix-sum` `monotonic-queue`

##### Approach 1: Prefix Sum + Monotonic Deque

For each right prefix, discard front prefixes that already form valid subarrays, taking the shortest first. Discard back prefixes no smaller than the current prefix because the current index is later and has a smaller or equal sum, so it dominates them for every future right boundary.

```python
from collections import deque


class Solution:
    def shortestSubarray(self, nums: list[int], k: int) -> int:
        prefix = [0]
        for num in nums:
            prefix.append(prefix[-1] + num)

        candidates = deque()
        answer = len(nums) + 1

        for right, current_sum in enumerate(prefix):
            while candidates and current_sum - prefix[candidates[0]] >= k:
                answer = min(answer, right - candidates.popleft())
            while candidates and prefix[candidates[-1]] >= current_sum:
                candidates.pop()
            candidates.append(right)

        return -1 if answer == len(nums) + 1 else answer
# Time: O(n), Space: O(n)
```

#### 548. [将数组分割成和相等的子数组](https://leetcode.com/problems/split-array-with-equal-sum/) [H]

Return whether three separator elements can leave four non-empty parts with equal sums.  
`array` `hash-table` `prefix-sum`

##### Approach 1: Prefix Sum + Middle Separator

Fix the middle separator `j`. Collect every sum achievable by equal first and second parts, then scan the third separator `k` for equal third and fourth parts with a sum in that set.

```python
class Solution:
    def splitArray(self, nums: list[int]) -> bool:
        n = len(nums)
        if n < 7:
            return False

        prefix = [0]
        for num in nums:
            prefix.append(prefix[-1] + num)

        for middle in range(3, n - 3):
            left_sums = set()
            for first_cut in range(1, middle - 1):
                first_sum = prefix[first_cut]
                second_sum = prefix[middle] - prefix[first_cut + 1]
                if first_sum == second_sum:
                    left_sums.add(first_sum)

            for third_cut in range(middle + 2, n - 1):
                third_sum = prefix[third_cut] - prefix[middle + 1]
                fourth_sum = prefix[n] - prefix[third_cut + 1]
                if third_sum == fourth_sum and third_sum in left_sums:
                    return True

        return False
# Time: O(n^2), Space: O(n)
```

#### 560. [和为 K 的子数组](https://leetcode.com/problems/subarray-sum-equals-k/) [M]

Count contiguous subarrays whose sum equals `k`.  
`array` `hash-table` `prefix-sum`

##### Approach 1: Prefix Sum Frequencies

If the current prefix sum is `prefix`, every earlier prefix equal to `prefix - k` starts a valid subarray. Store frequencies rather than only indices because multiple earlier prefixes can produce different subarrays.

```python
class Solution:
    def subarraySum(self, nums: list[int], k: int) -> int:
        count = {0: 1}
        prefix = answer = 0

        for num in nums:
            prefix += num
            answer += count.get(prefix - k, 0)
            count[prefix] = count.get(prefix, 0) + 1

        return answer
# Time: O(n), Space: O(n)
```

#### 209. [长度最小的子数组](https://leetcode.com/problems/minimum-size-subarray-sum/) [M]

Find the minimum length of a positive-integer subarray whose sum is at least `target`.  
`array` `binary-search` `prefix-sum` `sliding-window`

##### Approach 1: Prefix Sum + Binary Search

Positive numbers make the prefix-sum array strictly increasing. For every left boundary, binary-search the first prefix at least `prefix[left] + target`.

```python
from bisect import bisect_left

class Solution:
    def minSubArrayLen(self, target: int, nums: list[int]) -> int:
        prefix = [0]
        for num in nums:
            prefix.append(prefix[-1] + num)

        answer = len(nums) + 1
        for left in range(len(nums)):
            right = bisect_left(prefix, prefix[left] + target, left + 1)
            if right <= len(nums):
                answer = min(answer, right - left)

        return 0 if answer == len(nums) + 1 else answer
# Time: O(n log n), Space: O(n)
```

##### Approach 2: Sliding Window

Because all values are positive, expanding can only increase the sum and shrinking can only decrease it.

```python
class Solution:
    def minSubArrayLen(self, target: int, nums: list[int]) -> int:
        left = window_sum = 0
        answer = len(nums) + 1

        for right, num in enumerate(nums):
            window_sum += num
            while window_sum >= target:
                answer = min(answer, right - left + 1)
                window_sum -= nums[left]
                left += 1

        return 0 if answer == len(nums) + 1 else answer
# Time: O(n), Space: O(1)
```

#### 238. [除了自身以外数组的乘积](https://leetcode.com/problems/product-of-array-except-self/) [M]

Return the product of every element except `nums[i]` without division.  
`array` `prefix-sum`

##### Approach 1: Prefix and Suffix Products

Write the product to the left into the answer, then multiply in the product to the right during a reverse pass.

```python
class Solution:
    def productExceptSelf(self, nums: list[int]) -> list[int]:
        answer = [1] * len(nums)

        prefix_product = 1
        for i, num in enumerate(nums):
            answer[i] = prefix_product
            prefix_product *= num

        suffix_product = 1
        for i in range(len(nums) - 1, -1, -1):
            answer[i] *= suffix_product
            suffix_product *= nums[i]

        return answer
# Time: O(n), Space: O(1) auxiliary (excluding output)
```

#### 1004. [最大连续 1 的个数 III](https://leetcode.com/problems/max-consecutive-ones-iii/) [M]

Find the longest subarray containing at most `k` zeros.  
`array` `binary-search` `prefix-sum` `sliding-window`

##### Approach 1: Zero Prefix Sum + Binary Search

Prefix sums count zeros. For each right boundary, binary-search the earliest prefix whose removal leaves at most `k` zeros.

```python
from bisect import bisect_left

class Solution:
    def longestOnes(self, nums: list[int], k: int) -> int:
        zeros = [0]
        for num in nums:
            zeros.append(zeros[-1] + (num == 0))

        answer = 0
        for right in range(1, len(zeros)):
            left = bisect_left(zeros, zeros[right] - k, 0, right)
            answer = max(answer, right - left)
        return answer
# Time: O(n log n), Space: O(n)
```

##### Approach 2: Sliding Window

Keep at most `k` zeros in the window and move `left` only when that invariant is broken.

```python
class Solution:
    def longestOnes(self, nums: list[int], k: int) -> int:
        left = zeros = 0

        for right, num in enumerate(nums):
            zeros += num == 0
            if zeros > k:
                zeros -= nums[left] == 0
                left += 1

        return len(nums) - left
# Time: O(n), Space: O(1)
```

#### 525. [连续数组](https://leetcode.com/problems/contiguous-array/) [M]

Find the maximum length of a contiguous subarray with equal numbers of zeros and ones.  
`array` `hash-table` `prefix-sum`

##### Approach 1: Balanced Prefix Sum

Treat zero as `-1` and one as `+1`. Equal prefix balances at two positions mean the elements between them sum to zero; retain only the earliest index for each balance.

```python
class Solution:
    def findMaxLength(self, nums: list[int]) -> int:
        first_index = {0: -1}
        balance = answer = 0

        for i, num in enumerate(nums):
            balance += 1 if num == 1 else -1
            if balance in first_index:
                answer = max(answer, i - first_index[balance])
            else:
                first_index[balance] = i

        return answer
# Time: O(n), Space: O(n)
```

#### 1878. [矩阵中最大的三个菱形和](https://leetcode.com/problems/get-biggest-three-rhombus-sums-in-a-grid/) [M]

Return the three largest distinct sums of rhombus borders in the grid.  
`array` `math` `matrix` `prefix-sum`

##### Approach 1: Diagonal Prefix Sums

Precompute sums on both diagonal directions. Each of the four rhombus edges is then an O(1) range query; subtract the four corners once because adjacent edges count them twice.

```python
class Solution:
    def getBiggestThree(self, grid: list[list[int]]) -> list[int]:
        rows, cols = len(grid), len(grid[0])
        down_right = [[0] * (cols + 1) for _ in range(rows + 1)]
        down_left = [[0] * (cols + 1) for _ in range(rows + 1)]

        for r in range(rows):
            for c in range(cols):
                down_right[r + 1][c + 1] = down_right[r][c] + grid[r][c]
            for c in range(cols - 1, -1, -1):
                down_left[r + 1][c] = down_left[r][c + 1] + grid[r][c]

        def sum_down_right(r1: int, c1: int, r2: int, c2: int) -> int:
            return down_right[r2 + 1][c2 + 1] - down_right[r1][c1]

        def sum_down_left(r1: int, c1: int, r2: int, c2: int) -> int:
            return down_left[r2 + 1][c2] - down_left[r1][c1 + 1]

        best = set()
        for top_r in range(rows):
            for center_c in range(cols):
                best.add(grid[top_r][center_c])
                max_radius = min((rows - 1 - top_r) // 2, center_c, cols - 1 - center_c)
                for radius in range(1, max_radius + 1):
                    middle_r = top_r + radius
                    bottom_r = top_r + 2 * radius
                    left_c = center_c - radius
                    right_c = center_c + radius
                    border = (

                        - grid[middle_r][left_c]
                        - grid[middle_r][right_c]
                        - grid[bottom_r][center_c]
```

#### 713. [乘积小于 K 的子数组](https://leetcode.com/problems/subarray-product-less-than-k/) [M]

Count contiguous subarrays whose product is strictly less than `k`.  
`array` `binary-search` `prefix-sum` `sliding-window`

##### Approach 1: Prefix Products + Binary Search

For each right prefix product, binary-search the first left prefix satisfying `prefix[left] * k > prefix[right]`. This is exact, but prefix products can grow very large, so the sliding-window alternative is preferable in an interview.

```python
class Solution:
    def numSubarrayProductLessThanK(self, nums: list[int], k: int) -> int:
        if k <= 1:
            return 0

        prefix = [1]
        for num in nums:
            prefix.append(prefix[-1] * num)

        answer = 0
        for right in range(1, len(prefix)):
            lo, hi = 0, right
            while lo < hi:
                mid = (lo + hi) // 2
                if prefix[mid] * k > prefix[right]:
                    hi = mid
                else:
                    lo = mid + 1
            answer += right - lo
        return answer
# Time: O(n log n), Space: O(n)
```

##### Approach 2: Sliding Window

All values are positive, so shrink from the left until the product is valid. Every subarray ending at `right` and starting inside the final window is valid.

```python
class Solution:

        for right, num in enumerate(nums):
            product *= num
            while product >= k:
```

```python
        rows, cols = len(mat), len(mat[0])
        prefix = [[0] * (cols + 1) for _ in range(rows + 1)]

        for r in range(rows):
            for c in range(cols):
                prefix[r + 1][c + 1] = (
                    mat[r][c]
                    + prefix[r][c + 1]
                    + prefix[r + 1][c]
                    - prefix[r][c]

                bottom, right = min(rows - 1, r + k), min(cols - 1, c + k)
                answer[r][c] = (
                    prefix[bottom + 1][right + 1]
                    - prefix[top][right + 1]
                    - prefix[bottom + 1][left]
                    + prefix[top][left]
                )
        return answer
```

```python
            n = nums[i]
            if s * 2 + n == sm:

        left_sum = 0

        for i, num in enumerate(nums):
            if left_sum == total - left_sum - num:
                return i
            left_sum += num
        return -1
```

#### 2389. [和有限的最长子序列](https://leetcode.com/problems/longest-subsequence-with-limited-sum/) [E]

For each query, return the maximum subsequence length whose sum does not exceed the query.  
`array` `binary-search` `greedy` `sorting` `prefix-sum`

##### Approach 1: Sorted Prefix Sum + Binary Search

The cheapest subsequence of any length uses the smallest values. Sort, build cumulative costs, and binary-search how many fit within each budget.

```python
from bisect import bisect_right

class Solution:
    def answerQueries(self, nums: list[int], queries: list[int]) -> list[int]:
        nums.sort()
        prefix = []
        total = 0
        for num in nums:
            total += num
            prefix.append(total)

        return [bisect_right(prefix, query) for query in queries]
# Time: O(n log n + q log n), Space: O(n) excluding output
```

#### 2485. [找出中枢整数](https://leetcode.com/problems/find-the-pivot-integer/) [E]

Find `x` such that the sum from `1` to `x` equals the sum from `x` to `n`.  
`math` `prefix-sum`

##### Approach 1: Running Prefix Sum

Compare each prefix through `x` with the suffix starting at `x`, derived from the total sum.

```python
class Solution:
    def pivotInteger(self, n: int) -> int:
        total = n * (n + 1) // 2
        prefix = 0

        for x in range(1, n + 1):
            prefix += x
            if prefix == total - prefix + x:
                return x
        return -1
# Time: O(n), Space: O(1)
```

#### 2848. [与车相交的点](https://leetcode.com/problems/points-that-intersect-with-cars/) [E]

Count integer points covered by at least one inclusive car interval.  
`array` `hash-table` `prefix-sum`

##### Approach 1: Difference Array + Prefix Sum

Add coverage at each interval start and remove it immediately after the inclusive end. Count positions whose running coverage is positive.

```python
class Solution:
    def numberOfPoints(self, nums: list[list[int]]) -> int:
        farthest = max(end for _, end in nums)
        delta = [0] * (farthest + 2)

        for start, end in nums:
            delta[start] += 1
            delta[end + 1] -= 1

        coverage = answer = 0
        for change in delta:
            coverage += change
            answer += coverage > 0
        return answer
# Time: O(n + U), Space: O(U), where U is the farthest endpoint
```
